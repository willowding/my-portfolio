# 1. 切到项目根
cd C:\cursor\willowding-portfolio

# 2. 确认 git 可用
git --version

# 3. 初始化
git init
git branch -M main

# 4. 配用户身份（首次）
git config user.name "Willow Ding"
git config user.email "willowding@users.noreply.github.com"

# 5. 加全部文件（.gitignore 会过滤掉 .next / node_modules）
git add .
git status --short

# 6. 提交
git commit -m "Initial commit: bilingual portfolio"

# 7. 加远程 + 推送
git remote add origin https://github.com/willowding/portfolio.git
git push -u origin main
